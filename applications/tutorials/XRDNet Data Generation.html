def get_composition(structure):
    atoms = Atoms(aaa.get_atoms(conventional_structure))
    composition = {}
    for a in atoms:
        if a.symbol not in composition:
            composition[a.symbol] = 1
        else:
            composition[a.symbol] += 1

    return composition

def get_cell_params(structure):
    atoms = Atoms(aaa.get_atoms(structure))
    return tuple(atoms.cell.cellpar().tolist())
